> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Redesign the Home tab from a 2-column layout (match-list + details) into a 3-column FotMob-style layout: a league navigator (A-Division / NSL / SAFF Championship) on the left, a date-agnostic Upcoming/Finished schedule in the middle, and the existing match details panel on the right. Clicking a league swaps the left column to a standings table (A-Division, NSL) or a grouped fixtures list (SAFF Championship) and filters the middle schedule to that league.

**Architecture:** Standings are computed client-side from matches already fetched by `App.tsx` — no backend changes. NSL currently has 7 teams but zero matches, so this plan first seeds a full 21-match round-robin (reconstructed to reproduce the user's reference standings screenshot exactly) plus a handful of upcoming return-leg fixtures. Everything else is new/rewritten frontend components in `frontend/src/`.

**Tech Stack:** React 19 + TypeScript + Vite + Tailwind v4. Backend seeding via `services/ml-service/scripts/seed_db.py` (Python + pymongo). No test runner exists for this frontend (`npm run lint` = `tsc --noEmit` is the only automated check) — verification is `tsc --noEmit` + manual `npm run dev` checks, same convention as the prior `2026-07-25-unslop-frontend-redesign.md` plan in this repo.

**Interfaces:**

- [ ] **Step 1: Create the file**

```ts
import { Match, Team } from '../types';

export interface StandingRow {
  team: Team;
  played: number;
  won: number;
  drawn: number;
  lost: number;
  goalsFor: number;
  goalsAgainst: number;
  points: number;
  last5: ('W' | 'D' | 'L')[];
}

function matchTimestamp(m: Match): number {
  return new Date(`${m.date} ${m.time}`).getTime();
}

interface RowAccumulator {
  played: number;
  won: number;
  drawn: number;
  lost: number;
  goalsFor: number;
  goalsAgainst: number;
  points: number;
  results: { ts: number; outcome: 'W' | 'D' | 'L' }[];
}

export function computeStandings(matches: Match[], competitions: string[]): StandingRow[] {
  const inLeague = matches.filter((m) => competitions.includes(m.competition));
  const finished = inLeague.filter((m) => m.status === 'FINISHED');

  const teams = new Map<string, Team>();
  const rows = new Map<string, RowAccumulator>();

  function ensureRow(team: Team): RowAccumulator {
    if (!rows.has(team.id)) {
      teams.set(team.id, team);
      rows.set(team.id, {
        played: 0, won: 0, drawn: 0, lost: 0,
        goalsFor: 0, goalsAgainst: 0, points: 0,
        results: [],
      });
    }
    return rows.get(team.id)!;
  }

  // Register every team that appears in this league, even if it has no finished
  // matches yet, so it still shows up as an all-zero row instead of being hidden.
  for (const m of inLeague) {
    ensureRow(m.homeTeam);
    ensureRow(m.awayTeam);
  }

  for (const m of finished) {
    const home = ensureRow(m.homeTeam);
    const away = ensureRow(m.awayTeam);
    const hs = m.homeScore ?? 0;
    const as = m.awayScore ?? 0;
    const ts = matchTimestamp(m);

    home.played += 1;
    away.played += 1;
    home.goalsFor += hs;
    home.goalsAgainst += as;
    away.goalsFor += as;
    away.goalsAgainst += hs;

    if (hs > as) {
      home.won += 1;
      home.points += 3;
      home.results.push({ ts, outcome: 'W' });
      away.lost += 1;
      away.results.push({ ts, outcome: 'L' });
    } else if (as > hs) {
      away.won += 1;
      away.points += 3;
      away.results.push({ ts, outcome: 'W' });
      home.lost += 1;
      home.results.push({ ts, outcome: 'L' });
    } else {
      home.drawn += 1;
      home.points += 1;
      home.results.push({ ts, outcome: 'D' });
      away.drawn += 1;
      away.points += 1;
      away.results.push({ ts, outcome: 'D' });
    }
  }

  const result: StandingRow[] = [];
  for (const [teamId, row] of rows) {
    const team = teams.get(teamId)!;
    const last5 = row.results
      .slice()
      .sort((a, b) => a.ts - b.ts)
      .slice(-5)
      .map((r) => r.outcome);
    result.push({
      team,
      played: row.played,
      won: row.won,
      drawn: row.drawn,
      lost: row.lost,
      goalsFor: row.goalsFor,
      goalsAgainst: row.goalsAgainst,
      points: row.points,
      last5,
    });
  }

  result.sort((a, b) => {
    if (b.points !== a.points) return b.points - a.points;
    const gdA = a.goalsFor - a.goalsAgainst;
    const gdB = b.goalsFor - b.goalsAgainst;
    if (gdB !== gdA) return gdB - gdA;
    if (b.goalsFor !== a.goalsFor) return b.goalsFor - a.goalsFor;
    return a.team.name.localeCompare(b.team.name);
  });

  return result;
}
```

```tsx
import { Match } from '../types';

interface LeagueFixturesProps {
  matches: Match[];
  competitions: string[];
  selectedMatchId: string;
  onSelectMatch: (matchId: string) => void;
}

function matchTimestamp(m: Match): number {
  return new Date(`${m.date} ${m.time}`).getTime();
}

export default function LeagueFixtures({ matches, competitions, selectedMatchId, onSelectMatch }: LeagueFixturesProps) {
  const groups = competitions
    .map((comp) => ({
      competition: comp,
      matches: matches.filter((m) => m.competition === comp),
    }))
    .filter((g) => g.matches.length > 0);

  return (
    <div className="bg-[#151619] border border-slate-800/60 rounded-xl overflow-hidden shadow-xl">
      <div className="p-4 border-b border-[#202124] bg-slate-950">
        <h3 className="text-sm font-sans font-semibold text-slate-200 uppercase tracking-wider">SAFF Championship</h3>
        <p className="text-[10px] text-slate-500 mt-0.5">Nepal's fixtures by edition</p>
      </div>

      {groups.length === 0 ? (
        <div className="p-8 text-center text-xs text-slate-600">No fixtures recorded yet</div>
      ) : (
        <div className="divide-y divide-slate-900">
          {groups.map((group) => {
            const sorted = [...group.matches].sort((a, b) => {
              const aUpcoming = a.status !== 'FINISHED';
              const bUpcoming = b.status !== 'FINISHED';
              if (aUpcoming && bUpcoming) return matchTimestamp(a) - matchTimestamp(b);
              if (!aUpcoming && !bUpcoming) return matchTimestamp(b) - matchTimestamp(a);
              return aUpcoming ? -1 : 1;
            });
            return (
              <div key={group.competition} className="p-3">
                <h4 className="text-[10px] font-bold text-slate-500 uppercase tracking-wider mb-2 px-1">
                  {group.competition}
                </h4>
                <div className="space-y-1">
                  {sorted.map((m) => {
                    const isSelected = m.id === selectedMatchId;
                    const nepalIsHome = m.homeTeam.country === 'Nepal';
                    const opponent = nepalIsHome ? m.awayTeam : m.homeTeam;
                    return (
                      <button
                        key={m.id}
                        onClick={() => onSelectMatch(m.id)}
                        className={`w-full flex items-center justify-between gap-2 px-2 py-1.5 rounded-lg text-left transition cursor-pointer ${
                          isSelected ? 'bg-brand-500/15 border border-brand-500/40' : 'hover:bg-slate-900/60 border border-transparent'
                        }`}
                      >
                        <span className="text-[11px] text-slate-300 truncate">
                          {nepalIsHome ? 'vs' : '@'} {opponent.shorthand}
                        </span>
                        {m.status === 'FINISHED' ? (
                          <span className="text-[10px] font-mono font-bold text-slate-200 shrink-0">
                            {m.homeScore}-{m.awayScore}
                          </span>
                        ) : (
                          <span className="text-[10px] font-mono text-brand-400 shrink-0">{m.date}</span>
                        )}
                      </button>
                    );
                  })}
                </div>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}

                  matches={matches}
                  selectedMatchId={selectedMatchId}
                  onSelectMatch={handleSelectMatch}
                  selectedLeague={selectedLeague}
                />
              )}
            </div>
          )}
```

Note: below the `xl` breakpoint the grid is single-column (`grid-cols-1`), so `LeagueNav`, `MatchFeed`, then `MatchDetails` simply stack in that DOM order — this simplifies the design spec's illustrative 3-tier (`mobile`/`lg`/`xl`) breakpoint sketch down to 2 tiers (mobile-stacked / `xl`-3-column) to avoid a CSS `display: contents` responsive-regrouping hack for an intermediate tablet state that isn't essential to the feature.
